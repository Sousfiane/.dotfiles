# The desktop: Hyprland, the Quickshell shell, and everything they run.
# archinstall's Hyprland profile brings part of it; listing it all here means the install doesn't depend
# on that profile (packages already installed are skipped).

# ---------- Hyprland ----------
hyprland
hyprpm                       # plugins (autostart.lua: hyprpm reload)
hypridle                     # screens off when idle (a user service: steps/services.sh)
uwsm                         # session manager: apps are started with uwsm-app
xdg-desktop-portal-hyprland  # screen sharing, file pickers
qt5-wayland
qt6-wayland
sddm                         # login screen (enabled by steps/services.sh)

# ---------- Quickshell and what it calls ----------
quickshell
kitty                        # the terminal
xdg-terminal-exec            # opens the terminal: SUPER + Return, and Quickshell's floating terminal
clipvault-bin                # clipboard history: stored from autostart.lua, read by the panel
wl-clipboard                 # wl-paste / wl-copy, for the clipboard history
grim                         # the lock screen's capture of each screen
hyprshot                     # the screenshot panel (pulls slurp)
libpulse                     # pactl: the mixer moves app streams and sets card profiles with it
pacman-contrib               # checkupdates: the updates panel
nm-connection-editor         # the network panel's "Edit"
upower                       # device batteries the Bluetooth panel can't get from BlueZ
libsecret                    # secret-tool: the calendar's password, from the keyring
python                       # scripts/calendar.py
python-dbus                  # bt-agent (started by autostart.lua)
python-gobject               # bt-agent
curl                         # weather

# ---------- audio, network, Bluetooth (services: steps/services.sh) ----------
pipewire
pipewire-alsa
pipewire-jack
pipewire-pulse
wireplumber
gst-plugin-pipewire
networkmanager
bluez
bluez-utils

# ---------- session ----------
gnome-keyring
xdg-user-dirs
xdg-utils

# ---------- look ----------
ttf-jetbrains-mono-nerd      # the shell's font
noto-fonts
noto-fonts-cjk
noto-fonts-emoji
rose-pine-gtk-theme-full     # GTK themes and icons, all three variants (set by config/theme/apply.sh)
libnotify                    # notify-send: theme/apply.sh's "restart Spotify" notification
hyprqt6engine
darkly                       # Qt apps' widget style: rounded, modern (set by config/theme/files.sh)
unicode-emoji                # the emoji list the launcher searches (":")
greetd                       # the login screen, with the Quickshell greeter (install.sh greeter)
